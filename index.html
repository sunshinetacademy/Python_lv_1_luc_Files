المستوى الأول 🟢 فتح الملفات

📐 الشكل العام:

file = open("filename.txt", "mode") # العمليات على الملف file.close()

نستخدم دالة open() لفتح الملفات!

أوضاع فتح الملفات (Modes)

الوضع المعنى الاستخدام
"r" قراءة (Read) قراءة محتوى الملف فقط
"w" كتابة (Write) كتابة محتوى جديد (يمسح القديم)
"a" إضافة (Append) إضافة في نهاية الملف

⚠️ تحذير مهم:

وضع "w" يمسح كل المحتوى القديم ويكتب محتوى جديد!

استخدم "a" إذا أردت الإضافة بدون مسح! ➕

المستوى الثاني 🟡 قراءة الملفات

دوال القراءة

# فتح الملف للقراءة file = open("notes.txt", "r") # قراءة كل المحتوى content = file.read() print(content) # إغلاق الملف file.close()

📋 دوال القراءة المختلفة

file = open("notes.txt", "r") # قراءة سطر واحد line = file.readline() print(line) # قراءة كل الأسطر كقائمة lines = file.readlines() print(lines) file.close()
🎮 جربها بنفسك - محاكاة القراءة
# محاكاة قراءة ملف # في Python الحقيقي، نستخدم open() # ملف افتراضي (في الذاكرة) file_content = """مرحباً بك في Python! هذا ملف نصي تجريبي. يمكنك قراءة هذا المحتوى.""" print("📖 محتوى الملف:") print("=" * 40) print(file_content) print("=" * 40) # تقسيم إلى أسطر lines = file_content.split('\n') print(f"\n عدد الأسطر: {len(lines)}") # قراءة سطر بسطر print("\n الأسطر:") for i, line in enumerate(lines, 1): print(f"سطر {i}: {line}")

المستوى الثالث الكتابة والإضافة

✍️ الكتابة في ملف (Write)

# فتح الملف للكتابة (يمسح المحتوى القديم) file = open("notes.txt", "w") # كتابة محتوى file.write("مرحباً بك في Python!\n") file.write("هذا أول ملف لي.\n") # إغلاق الملف file.close() print("✅ تم الكتابة بنجاح!")

➕ الإضافة في نهاية الملف (Append)

# فتح الملف للإضافة (لا يمسح المحتوى) file = open("notes.txt", "a") # إضافة محتوى جديد file.write("هذا سطر جديد أُضيف في النهاية.\n") file.close() print("✅ تمت الإضافة بنجاح!")
🎮 جربها بنفسك - محاكاة الكتابة
# محاكاة الكتابة والإضافة # ملف في الذاكرة file_content = [] print("📝 نظام المذكرات") print("=" * 40) # كتابة محتوى جديد print("\n✍️ كتابة محتوى جديد...") file_content = ["مرحباً!", "هذا أول ملف لي."] print(f"تم كتابة {len(file_content)} سطر") # إضافة محتوى print("\n إضافة محتوى...") file_content.append("هذا سطر جديد!") print(f"الآن الملف يحتوي {len(file_content)} سطر") # عرض المحتوى print("\n محتوى الملف:") for i, line in enumerate(file_content, 1): print(f"{i}. {line}")
اضغط Run ▶

💡 الفرق بين Write و Append:

  • "w" = يمسح كل شيء ويكتب جديد (مثل صفحة بيضاء)
  • "a" = يضيف في النهاية (مثل كتابة في آخر الصفحة)

المستوى الرابع 🟣 الطريقة الآمنة (with)

🌟 لماذا نستخدم with؟

عند استخدام open() العادية، يجب أن نتذكر إغلاق الملف بـ close()!

لو نسينا، قد يحدث مشاكل في الذاكرة! 😰

الحل: استخدام with - يغلق الملف تلقائياً! ✨

🔐 الشكل الآمن

# الطريقة الآمنة - تغلق الملف تلقائياً! with open("notes.txt", "r") as file: content = file.read() print(content) # الملف يُغلق تلقائياً هنا!

📝 أمثلة عملية

# قراءة ملف بأمان with open("notes.txt", "r") as file: content = file.read() print("📖 المحتوى:", content) # كتابة ملف بأمان with open("notes.txt", "w") as file: file.write("محتوى جديد!\n") print("✅ تم الكتابة") # إضافة بأمان with open("notes.txt", "a") as file: file.write("سطر جديد!\n") print("✅ تمت الإضافة")
🎮 جربها بنفسك - النظام الكامل
# نظام المذكرات الكامل print("📔 نظام المذكرات الشخصية") print("=" * 40) # محاكاة ملف في الذاكرة diary = [] # إضافة ملاحظات notes = [ "اليوم كان يوماً رائعاً!", "تعلمت Python اليوم 🐍", "الملفات سهلة جداً!" ] print("\n✍️ كتابة ملاحظات جديدة...") for note in notes: diary.append(note) print(f" ✅ {note}") # قراءة الملاحظات print("\n📖 قراءة المذكرة:") print("-" * 40) for i, note in enumerate(diary, 1): print(f"{i}. {note}") print("-" * 40) print(f"📊 إجمالي الملاحظات: {len(diary)}")

🎨 المشروع: نظام المذكرات الشخصية

هنعمل نظام كامل لحفظ وقراءة المذكرات الشخصية!

📝 نظام المذكرات

🎮 جرب المشروع
# ═══════════════════════════════════════ # 📔 نظام المذكرات الشخصية # ═══════════════════════════════════════ print("📔 مرحباً بك في نظام المذكرات!") print("=" * 40) # محاكاة ملف المذكرة diary_file = [] # قائمة الخيارات while True: print("\n القائمة:") print("1. ️ كتابة ملاحظة جديدة") print("2. 📖 قراءة كل الملاحظات") print("3. البحث في الملاحظات") print("4. 📊 إحصائيات") print("5. 🚪 الخروج") choice = "1" # في الواقع: input("اختر: ") if choice == "1": print("\n✍️ اكتب ملاحظتك:") note = "تعلمت Python اليوم! 🐍" diary_file.append(note) print(f"✅ تم حفظ: {note}") elif choice == "2": print("\n📖 ملاحظاتك:") print("-" * 40) if diary_file: for i, note in enumerate(diary_file, 1): print(f"{i}. {note}") else: print("لا توجد ملاحظات بعد!") print("-" * 40) elif choice == "3": print("\n🔍 ابحث عن كلمة:") keyword = "Python" print(f"\nنتائج البحث عن '{keyword}':") found = False for note in diary_file: if keyword in note: print(f" ✅ {note}") found = True if not found: print(" ❌ لم يتم العثور على نتائج") elif choice == "4": print("\n📊 إحصائيات:") print(f" 📝 عدد الملاحظات: {len(diary_file)}") total_chars = sum(len(note) for note in diary_file) print(f" 🔤 إجمالي الأحرف: {total_chars}") elif choice == "5": print("\n👋 وداعاً! تم حفظ ملاحظاتك.") break print("\n" + "=" * 40) print("🎉 شكرًا لاستخدام نظام المذكرات!")
